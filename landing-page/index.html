Get Started in 60 Seconds

// 1. Install CLI
npx @iam-system/cli init --framework react

// 2. Wrap your app
import { IamProvider } from '@iam-system/react';

function App() {
  return (
    <IamProvider domain="your-domain.iam.dev" clientId="your-client-id">
      <YourApp />
    </IamProvider>
  );
}

// 3. Add login
import { useAuth } from '@iam-system/react';

function LoginButton() {
  const { loginWithPasskey } = useAuth();
  return <button onClick={loginWithPasskey}>Login with Face ID</button>;
}

// That's it! ✅
// 1. Install CLI
npx @iam-system/cli init --framework vue

// 2. Setup in main.ts
import { createIam } from '@iam-system/vue';

app.use(createIam({
  domain: 'your-domain.iam.dev',
  clientId: 'your-client-id'
}));

// 3. Use in component
import { useAuth } from '@iam-system/vue';

const { loginWithPasskey } = useAuth();

// That's it! ✅
// 1. Install CLI
npx @iam-system/cli init --framework nextjs

// 2. Add API route (app/api/auth/[...iam]/route.ts)
import { IamAuth } from '@iam-system/nextjs';

export const { GET, POST } = IamAuth({
  domain: process.env.IAM_DOMAIN,
  clientId: process.env.IAM_CLIENT_ID
});

// 3. Use in component
import { useAuth } from '@iam-system/nextjs';

export default function LoginButton() {
  const { loginWithPasskey } = useAuth();
  return <button onClick={loginWithPasskey}>Login</button>;
}

// That's it! ✅
// 1. Install CLI
npx @iam-system/cli init --framework angular

// 2. Add to app.config.ts
import { provideIam } from '@iam-system/angular';

export const appConfig = {
  providers: [
    provideIam({
      domain: 'your-domain.iam.dev',
      clientId: 'your-client-id'
    })
  ]
};

// 3. Use in component
import { IamService } from '@iam-system/angular';

export class LoginComponent {
  constructor(private iam: IamService) {}

  login() {
    this.iam.loginWithPasskey();
  }
}

// That's it! ✅
